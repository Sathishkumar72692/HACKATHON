            get(collection) { return JSON.parse(localStorage.getItem(`ss_${collection}`) || '[]'); }
            set(collection, data) { localStorage.setItem(`ss_${collection}`, JSON.stringify(data)); }
            insert(collection, item) {
                const data = this.get(collection);
                item.id = `${collection.substring(0,3).toUpperCase()}-${Math.floor(1000 + Math.random() * 9000)}`;
                item.createdAt = new Date().toISOString();
                data.push(item);
                this.set(collection, data);
                return item;
            }
            update(collection, id, updates) {
                let data = this.get(collection);
                const index = data.findIndex(i => i.id === id);
                if(index > -1) {
                    data[index] = { ...data[index], ...updates, updatedAt: new Date().toISOString() };
                    this.set(collection, data);
                    return data[index];
                }
                return null;
            }

                await this.delay(300);

                e.preventDefault();
                const mode = document.getElementById('auth-mode').value;
                const role = document.getElementById('auth-role').value;
                const email = document.getElementById('auth-email').value;
                const password = document.getElementById('auth-password').value;

                const user = API.getSession();

                const problems = await API.getProblems({ authorId: user.id });

                document.getElementById('cit-stat-progress').innerText = problems.filter(p => ['IN_PROGRESS', 'SOLUTION_PROPOSED'].includes(p.status)).length;
                document.getElementById('cit-stat-resolved').innerText = problems.filter(p => p.status === 'RESOLVED').length;

                const list = document.getElementById('citizen-reports-list');

                    list.innerHTML = `<tr><td colspan="6" class="px-6 py-8 text-center text-gray-500">You haven't reported any problems yet.</td></tr>`;
                    return;

                list.innerHTML = problems.map(p => {

                        'SUBMITTED': 'text-gray-400 bg-gray-800',

                        'RESOLVED': 'text-green-400 bg-green-900/30 border border-green-800'

                    const colorClass = statusColors[p.status] || 'text-gray-400 bg-gray-800';
                    return `

                    return;

                    // Add marker

                    marker.bindPopup(`<b class="text-slate-900">${p.title}</b><br><span class="text-slate-600">${p.category}</span>`);
                    
                    // Add list item
                    const severityColor = p.severity === 'Critical' ? 'text-red-400' : (p.severity === 'High' ? 'text-orange-400' : 'text-yellow-400');
                    list.innerHTML += `
                        <div class="bg-slate-800/50 p-4 rounded-lg border border-gray-700/50 hover:border-electric-500/50 transition-colors cursor-pointer" onclick="app.panTo(${p.lat}, ${p.lng})">
                            <div class="flex justify-between items-start mb-2">
                                <h3 class="font-bold text-white text-sm line-clamp-2">${p.title}</h3>
                                <span class="px-2 py-0.5 rounded text-[10px] uppercase font-bold bg-gray-900 ${severityColor}">${p.severity}</span>
                            </div>
                            <p class="text-xs text-gray-400 line-clamp-2 mb-3">${p.description}</p>
                            <div class="flex justify-between items-center text-xs">
                                <span class="text-cyan-400"><i class="fa-solid fa-tag mr-1"></i> ${p.category}</span>
                                <span class="text-gray-500">${p.status.replace('_', ' ')}</span>
                            </div>
                        </div>
                    `;
                });
            },
            
            panTo(lat, lng) {
                this.exploreMapInstance.flyTo([lat, lng], 12, { duration: 1.5 });
            }
        };

        const Router = {
            routes: ['home', 'auth', 'dashboard-citizen', 'dashboard-university', 'dashboard-industry', 'dashboard-government', 'report-problem', 'explore'],
            init: () => {
                window.addEventListener('hashchange', Router.handleRoute);
                Router.handleRoute();
            },
            navigate: (path) => {
                window.location.hash = path;
            },
            handleRoute: () => {
                let path = window.location.hash.substring(1);
                if (!path || !Router.routes.includes(path)) path = 'home';
                
                // Auth guards
                const user = API.getSession();
                if (path.startsWith('dashboard-') && (!user || `dashboard-${user.role}` !== path)) {
                    return Router.navigate('auth');
                }
                if (path === 'report-problem' && (!user || user.role !== 'citizen')) {
                    UI.showToast('Only citizens can report problems.', 'error');
                    return Router.navigate('auth');
                }
                
                const root = document.getElementById('app-root');
                root.innerHTML = Views[path]();

                // Post-render hooks
                if (path === 'dashboard-citizen') app.loadCitizenDashboard();
                if (path === 'report-problem') setTimeout(() => app.nextStep(1), 10);
                if (path === 'explore') setTimeout(() => app.initExplore(), 100);
                if (path === 'home') {
                    // Quick stat animation
                    const el = document.getElementById('stat-problems');
                    if(el) {
                        const total = DB.get('problems').length;
                        el.innerText = total;
                    }
                }
                
                window.scrollTo(0,0);
            }
        };

        window.onload = app.init;

    </script>
